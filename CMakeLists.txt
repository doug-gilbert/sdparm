cmake_minimum_required ( VERSION 3.18 )

project(sdparm VERSION 1.12.9)
# Use version number: x.y.9 to indicate pre-release of x.(y+1)
# Example: VERSION 1.12.9 is pre-release of 1.13

set ( CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -O2 -g" )

option ( BUILD_SHARED_LIBS "Build using shared libraries" ON)
option ( ENABLE_DEBUG "Turn on debugging (extra warnings)" OFF)
option ( DISABLE_FAST_LEBE "Use generic little-endian/big-endian code instead" OFF)
option ( DISABLE_SCSISTRINGS "Save space in executable using numeric error codes instead of strings" OFF)
option ( DISABLE_LINUXBSG "Remove support for /dev/bsg/* devices" OFF)
option ( DISABLE_LIBSGUTILS "Same as -DBUILD_SHARED_LIBS=OFF" OFF)
option ( DISABLE_NVME_SUPP "Remove all or most NVMe code" OFF)

include ( CMakePackageConfigHelpers )
include ( GNUInstallDirs )
include ( CheckIncludeFile )
include ( CheckFunctionExists )
include ( CheckCSourceCompiles )
find_package ( PkgConfig )

string(TIMESTAMP BUILD_TIME "%Y%m%d %H:%M:%S")

CHECK_INCLUDE_FILE( "linux/nvme_ioctl.h" HAVE_NVME )
CHECK_INCLUDE_FILE( "byteswap.h" HAVE_BYTESWAP_H )
CHECK_INCLUDE_FILE( "linux/major.h" HAVE_LINUX_MAJOR_H )

check_function_exists( getopt_long HAVE_GETOPT_LONG )
check_function_exists( posix_memalign HAVE_POSIX_MEMALIGN )
check_function_exists( posix_fadvise HAVE_POSIX_FADVISE )
check_function_exists( sysconf HAVE_SYSCONF )
check_function_exists( lseek64 HAVE_LSEEK64 )

if (DISABLE_LIBSGUTILS)
    set(BUILD_SHARED_LIBS OFF)
endif ()

set(FEATURE_COMMENT "//")

if ( ENABLE_DEBUG )
    MESSAGE( STATUS ">> Compile with DEBUG set" )
endif ( ENABLE_DEBUG )

set(IGNORE_FAST_LEBE 0)
if (DISABLE_FAST_LEBE)
    set(IGNORE_FAST_LEBE 1)
endif (DISABLE_FAST_LEBE)

set(IGNORE_LINUX_BSG 0)
if (DISABLE_LINUXBSG)
    set(IGNORE_LINUX_BSG 1)
endif ()

set(SG_SCSI_STRINGS 1)
if (DISABLE_SCSISTRINGS)
    set(SG_SCSI_STRINGS 0)
endif()

set(IGNORE_NVME 0)
if (DISABLE_NVME_SUPP)
    set(IGNORE_NVME 1)
endif()

set(OS_LINUX 0)
set(SG_LIB_LINUX 0)
set(OS_ANDROID 0)
set(OS_FREEBSD 0)
set(SG_LIB_FREEBSD 0)
set(OS_SOLARIS 0)
set(SG_LIB_SOLARIS 0)
set(OS_AIX 0)
set(OS_WINDOWS 0)
set(OS_CYGWIN 0)
set(OS_MINGW 0)

if(CYGWIN)
    message(STATUS "Cygwin")
    set(OS_CYGWIN 1)
    set(SG_LIB_WIN32 1)
    set(HAVE_NVME 1)
elseif(MINGW)
    message(STATUS "MinGW")
    set(OS_MINGW 1)
    set(SG_LIB_WIN32 1)
    set(SG_LIB_MINGW 1)
    set(HAVE_NVME 1)
elseif(WIN32)
    message(STATUS "Windows")
    set(OS_WINDOWS 1)
    set(SG_LIB_WIN32 1)
elseif(ANDROID)
    message(STATUS "Android")
    set(OS_ANDROID 1)
    set(SG_LIB_ANDROID 1)
    set(SG_LIB_LINUX 1)
    CHECK_INCLUDE_FILE( "linux/types.h" HAVE_LINUX_TYPES_H )
    CHECK_INCLUDE_FILE( "linux/bsg.h" HAVE_LINUX_BSG_H )
    CHECK_INCLUDE_FILE( "linux/kdev_t.h" HAVE_LINUX_KDEV_T_H )
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(STATUS "Linux")
    set(OS_LINUX 1)
    set(SG_LIB_LINUX 1)
    CHECK_INCLUDE_FILE( "linux/types.h" HAVE_LINUX_TYPES_H )
    CHECK_INCLUDE_FILE( "linux/bsg.h" HAVE_LINUX_BSG_H )
    CHECK_INCLUDE_FILE( "linux/kdev_t.h" HAVE_LINUX_KDEV_T_H )
elseif(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
    message(STATUS "FreeBSD")
    set(OS_FREEBSD 1)
    set(SG_LIB_FREEBSD 1)
    set(HAVE_NVME 1)
elseif(CMAKE_SYSTEM_NAME STREQUAL "NetBSD")
    message(STATUS "NetBSD")
    set(OS_NETBSD 1)
    set(SG_LIB_NETBSD 1)
elseif(CMAKE_SYSTEM_NAME STREQUAL "OpenBSD")
    message(STATUS "OpenBSD")
    set(OS_OPENBSD 1)
    set(SG_LIB_OPENBSD 1)
elseif(CMAKE_SYSTEM_NAME STREQUAL "SunOS")
    message(STATUS "Solaris")
    set(OS_SOLARIS 1)
    set(SG_LIB_SOLARIS 1)
elseif(CMAKE_SYSTEM_NAME STREQUAL "AIX")
    message(STATUS "AIX")
    set(OS_AIX 1)
elseif(APPLE)
    message(STATUS "macOS")
    set(OS_MACOS 1)
    set(SG_LIB_MACOS 1)
else()
    message(STATUS "Unknown system: ${CMAKE_SYSTEM_NAME}")
endif()

if (HAVE_LINUX_TYPES_H)
    check_c_source_compiles("
        #include <linux/types.h>
        int main(void) { __u64 var = 0; return (int)var; }
    " HAVE___U64)
endif ()

string(TOLOWER "${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}" SG_LIB_BUILD_HOST)

configure_file( ${CMAKE_SOURCE_DIR}/config.h.in.cmake ${CMAKE_BINARY_DIR}/config.h )
add_definitions ( -DHAVE_CONFIG_H )


add_subdirectory( src )

add_subdirectory( doc )

install(FILES ChangeLog DESTINATION "${CMAKE_INSTALL_DOCDIR}")
install(FILES README DESTINATION "${CMAKE_INSTALL_DOCDIR}")
install(FILES AUTHORS DESTINATION "${CMAKE_INSTALL_DOCDIR}")
install(FILES CREDITS DESTINATION "${CMAKE_INSTALL_DOCDIR}")

install(PROGRAMS scripts/sas_disk_blink DESTINATION "${CMAKE_INSTALL_BINDIR}")
if ( OS_LINUX )
    install(PROGRAMS scripts/scsi_ch_swp DESTINATION "${CMAKE_INSTALL_BINDIR}")
endif ()


set(CPACK_PROJECT_NAME ${PROJECT_NAME})
set(CPACK_PROJECT_VERSION "${PROJECT_VERSION}")

set(CPACK_GENERATOR "")
find_program(DPKG_EXECUTABLE dpkg)
if (DPKG_EXECUTABLE)
    list(APPEND CPACK_GENERATOR "DEB")
endif ()
find_program(RPMBUILD_EXECUTABLE rpmbuild)
if (RPMBUILD_EXECUTABLE)
    list(APPEND CPACK_GENERATOR "RPM")
endif ()

SET(CPACK_DEBIAN_PACKAGE_MAINTAINER "Douglas Gilbert")
SET(CPACK_DEBIAN_PACKAGE_SECTION "admin")
SET(CPACK_RPM_PACKAGE_DESCRIPTION "Permits reading and setting of SCSI device mode pages (cf hdparm for ATA disks)")
SET(CPACK_RPM_PACKAGE_LICENSE "BSD-2-clause")
SET(CPACK_RPM_PACKAGE_VENDOR "Douglas Gilbert")
if ( BUILD_SHARED_LIBS )
    SET(CPACK_RPM_PACKAGE_REQUIRES "sg3_utils-libs")
    SET(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
endif ()

include(CPack)
include(CTest)
